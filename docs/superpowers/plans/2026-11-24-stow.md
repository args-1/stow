> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rename Linkstash to Stow (keeping users' local data), show saved links as square cards in a responsive grid, and give every category an icon drawn in its colour instead of the colour dot.

**Tech Stack:** pnpm workspaces, TypeScript 7 (strict), React 19, Vite 8, Tailwind CSS 4, Dexie 4.4 + fake-indexeddb, zod 4, supabase-js 2, lucide-react 1.78, WXT 0.21 (MV3), Vitest 5, Testing Library, PGlite.

**Spec:** `docs/superpowers/specs/2026-11-24-stow-rename-cards-icons-design.md` (read it before starting; sections are cited as §n).

```ts
  'folder', 'briefcase', 'house', 'book-open', 'graduation-cap', 'code', 'terminal', 'newspaper',
  'video', 'film', 'music', 'image', 'camera', 'gamepad-2', 'shopping-cart', 'wallet',

export type CategoryIconName = (typeof CATEGORY_ICONS)[number];

export const DEFAULT_CATEGORY_ICON: CategoryIconName = 'folder';

export function isCategoryIcon(value: unknown): value is CategoryIconName {
  return typeof value === 'string' && (CATEGORY_ICONS as readonly string[]).includes(value);
}

/** The icon to draw or edit: `value` when it is a known name, otherwise DEFAULT_CATEGORY_ICON. */

  return isCategoryIcon(value) ? value : DEFAULT_CATEGORY_ICON;
}
```

```ts
export interface CategoryInput {
  name: string;
  color: string;
  /** One of CATEGORY_ICONS. */
  icon: string;
}

const categoryInputSchema = z.object({
  name: z
    .string()
    .trim()
    .min(1, 'Name is required')
    .max(LIMITS.categoryName, `Name must be at most ${LIMITS.categoryName} characters`),
  color: z.string().regex(/^#[0-9a-f]{6}$/i, 'Pick a color'),
  icon: z.custom<CategoryIconName>(isCategoryIcon, 'Pick an icon'),
});

export function parseCategoryInput(input: CategoryInput): Result<CategoryInput> {
  const parsed = categoryInputSchema.safeParse(input);
  if (!parsed.success) return { ok: false, error: parsed.error.issues[0]?.message ?? 'Invalid category' };
  return {
    ok: true,
    value: { name: parsed.data.name, color: parsed.data.color.toLowerCase(), icon: parsed.data.icon },
  };
}

/** Pass input that already went through parseCategoryInput. */
export function createCategory(input: CategoryInput, clock: Clock = systemClock): Category {
  const now = clock.now();
  return {
    id: clock.id(),
    name: input.name,
    color: input.color,
    icon: input.icon,
    created_at: now,
    updated_at: now,
    deleted_at: null,
  };
}

export type CategoryPatch = Partial<Pick<Category, 'name' | 'color' | 'icon' | 'deleted_at'>>;
```

```ts
export class RemoteError extends Error {
  /** HTTP status, or 0 when the request never reached the server. */
  readonly status: number;
  /** PostgREST or Postgres error code, when the server sent one. */
  readonly code?: string;

  constructor(message: string, status: number, code?: string) {
    super(message);
    this.name = 'RemoteError';
    this.status = status;
    this.code = code;
  }
}

export type ErrorKind = 'auth' | 'retryable' | 'fatal';

/** PostgREST's "column not in the schema cache": the client is newer than the database's migrations. */
const MISSING_COLUMN = 'PGRST204';

export function errorKind(error: unknown): ErrorKind {
  if (!(error instanceof RemoteError)) return 'retryable';
  if (error.code === MISSING_COLUMN) return 'retryable';
  const { status } = error;
  if (status === 401) return 'auth';
  if (status === 0 || status === 408 || status === 429 || status >= 500) return 'retryable';
  return 'fatal';
}
```

```ts
export const DB_NAME = 'stow';
/** The database name before the app was renamed to Stow. Its data is imported once, then it is deleted. */
export const LEGACY_DB_NAME = 'linkstash';

/**
 * Opens a database. When `legacyName` is set, opening first copies that database in (see importLegacyDb);
 * Dexie holds every other request on the returned db until the copy has finished.
 */
export function openDb(
  name = DB_NAME,
  legacyName: string | null = name === DB_NAME ? LEGACY_DB_NAME : null,
): StowDb {
  const db = new Dexie(name) as StowDb;
  db.version(1).stores({
    links: 'id',
    categories: 'id',
    outbox: '++seq',
    meta: 'key',
  });
  if (legacyName) db.on('ready', (vipDb) => importLegacyDb(vipDb as StowDb, legacyName));
  return db;
}

/** Categories saved before category icons existed have no icon. */
const withDefaultIcon = (category: Category): Category => ({
  ...category,
  icon: category.icon ?? DEFAULT_CATEGORY_ICON,
});

const withIcon = (mutation: Mutation): Mutation =>
  mutation.type === 'upsertCategory' ? { ...mutation, row: withDefaultIcon(mutation.row) } : mutation;

/**
 * Copies `legacyName` into `db` once, then deletes it. Copies nothing, and keeps the legacy database, when `db`
 * already has data. Runs in Dexie's ready hook, so `db` must be the hook's vipDb: a query through the outer db
 * would wait for this hook and deadlock. If this throws, opening fails and the import runs again next time.
 */
async function importLegacyDb(db: StowDb, legacyName: string): Promise<void> {
  if (!(await Dexie.exists(legacyName))) return;

  const legacy = openDb(legacyName, null);
  let rows: [Link[], Category[], OutboxEntry[], MetaRow[]];
  try {
    rows = await Promise.all([
      legacy.links.toArray(),
      legacy.categories.toArray(),
      legacy.outbox.orderBy('seq').toArray(),
      legacy.meta.toArray(),
    ]);
  } finally {
    legacy.close();
  }
  const [links, categories, outbox, meta] = rows;

  const empty = links.length + categories.length + outbox.length + meta.length === 0;
  // The emptiness check and the copy share one transaction, so a second tab importing at the same time copies nothing.
  const copied =
    !empty &&
    (await db.transaction('rw', db.tables, async () => {
      const counts = await Promise.all(db.tables.map((table) => table.count()));
      if (counts.some((count) => count > 0)) return false;
      await db.links.bulkAdd(links);
      await db.categories.bulkAdd(categories.map(withDefaultIcon));
      // Original seq values keep the push order of unsynced changes.
      await db.outbox.bulkAdd(outbox.map((entry) => ({ ...entry, mutation: withIcon(entry.mutation) })));
      await db.meta.bulkAdd(meta);
      return true;
    }));

  if (empty || copied) {
    // Not awaited: a tab still running the old build can block the delete, and start-up must not wait for it.
    void Dexie.delete(legacyName).catch((error: unknown) =>
      console.warn('Could not delete the old Linkstash database', error),
    );
  }
}
```

```tsx
  onToggleArchive(): void;
  onEdit(): void;
  onDelete(): void;
}

export function LinkCard({ link, category, onOpen, onToggleFavorite, onToggleArchive, onEdit, onDelete }: LinkCardProps) {
  const fallbackIcon = faviconFallback(link.url);

  return (
    <li
      className={
        'relative flex aspect-square flex-col gap-2 overflow-hidden rounded-xl border border-slate-200 bg-white p-2.5 ' +
        'shadow-sm has-[a:focus-visible]:ring-2 has-[a:focus-visible]:ring-indigo-500/50 dark:border-slate-800 dark:bg-slate-900'
      }
    >
      <div className="flex items-center gap-1.5 text-xs text-slate-500 dark:text-slate-400">
        <img
          src={link.favicon_url ?? fallbackIcon}
          alt=""
          width={24}
          height={24}
          loading="lazy"
          referrerPolicy="no-referrer"
          className="size-6 shrink-0 rounded"
          onError={(event) => {
            if (event.currentTarget.src !== fallbackIcon) event.currentTarget.src = fallbackIcon;
          }}
        />
        <span className="min-w-0 flex-1 truncate">{hostname(link.url)}</span>
        {category && (
          <span role="img" aria-label={`Category: ${category.name}`} title={category.name} className="relative z-10 shrink-0">
            <CategoryIcon icon={category.icon} color={category.color} />
          </span>
        )}
      </div>
      {/* Stretched link: its ::after covers the card, so tapping anywhere outside the buttons opens it. */}
      <a
        href={link.url}
        target="_blank"
        rel="noopener noreferrer"
        onClick={onOpen}
        onAuxClick={(event) => {
          if (event.button === 1) onOpen();
        }}
        className={
          'line-clamp-3 text-sm font-medium break-words after:absolute after:inset-0 hover:text-indigo-600 ' +
          'focus-visible:outline-none dark:hover:text-indigo-400'
        }
      >
        {displayTitle(link)}
      </a>
      <div className="relative z-10 mt-auto flex items-center justify-between">
        <IconButton size="sm" label="Favorite" pressed={link.is_favorite} onClick={onToggleFavorite}>
          <Star className={`size-4 ${link.is_favorite ? 'fill-amber-400 text-amber-400' : ''}`} />
        </IconButton>
        <IconButton size="sm" label={link.is_archived ? 'Unarchive' : 'Archive'} onClick={onToggleArchive}>
          {link.is_archived ? <ArchiveRestore className="size-4" /> : <Archive className="size-4" />}
        </IconButton>
        <IconButton size="sm" label="Edit" onClick={onEdit}>
          <Pencil className="size-4" />
        </IconButton>
        <IconButton size="sm" label="Delete" onClick={onDelete} className="hover:text-red-600 dark:hover:text-red-400">
          <Trash2 className="size-4" />
        </IconButton>
      </div>
    </li>
  );
}
```

Why this works:
- The `<a>` is not positioned, so its `::after` (`absolute inset-0`) is placed against the `relative` `<li>` and covers the whole card. `line-clamp-3` gives the `<a>` `overflow: hidden`, but that does not clip the `::after`, because its containing block is the `<li>`, outside the `<a>`.
- The category badge and the action row are `relative z-10`, so they sit above the overlay: the buttons stay clickable and the badge's tooltip works.
- The `<a>` hides its own outline; the `<li>` shows the ring through `has-[a:focus-visible]`.
